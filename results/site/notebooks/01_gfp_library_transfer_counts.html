Robust GFP Library Transfer Counts

Purpose

This notebook replaces the exploratory GFP library transfer-count analysis with a reproducible workflow that:

  • validates all required transfer-count inputs up front,
  • standardizes the eight GFP library conditions into one schema,
  • exports cleaned tables under results/01_gfp_library_transfer_counts/,
  • regenerates the main heatmap, kinetics, and viability plots.

resolve_project_root <- function() {
  rstudio_path <- ""
  if (requireNamespace("rstudioapi", quietly = TRUE)) {
    rstudio_path <- tryCatch(rstudioapi::getSourceEditorContext()$path, error = function(e) "")
    if (!nzchar(rstudio_path)) {
      rstudio_path <- tryCatch(rstudioapi::getActiveDocumentContext()$path, error = function(e) "")
    }
  }

  search_roots <- unique(c(
    Sys.getenv("CAS12A_NOTEBOOK_PROJECT_ROOT", unset = ""),
    tryCatch(knitr::current_input(dir = TRUE), error = function(e) ""),
    rstudio_path,
    getwd(),
    normalizePath(file.path(getwd(), ".."), mustWork = FALSE),
    normalizePath(file.path(getwd(), "..", ".."), mustWork = FALSE)
  ))
  search_roots <- search_roots[nzchar(search_roots)]

  for (root in search_roots) {
    current <- normalizePath(root, mustWork = FALSE)
    if (!dir.exists(current)) {
      current <- dirname(current)
    }

    repeat {
      required_paths <- c(
        file.path(current, "_quarto.yml"),
        file.path(current, "config", "paths.R"),
        file.path(current, "R", "common.R")
      )

      if (all(file.exists(required_paths))) {
        return(current)
      }

      parent <- dirname(current)
      if (identical(parent, current)) {
        break
      }

      current <- parent
    }
  }

  stop(
    paste(
      "Could not locate the notebook project root.",
      "Save the notebook and either run it from the project root,",
      "or set Sys.setenv(CAS12A_NOTEBOOK_PROJECT_ROOT = '/path/to/New project')."
    ),
    call. = FALSE
  )
}

plot_first_timepoint_viability <- function(data, cas_variant_name) {
  plot_data <- data %>%
    dplyr::filter(cas_variant == cas_variant_name) %>%
    dplyr::mutate(donor_recruitment = label_recruitment(donor_recruitment))

  ggplot(plot_data, aes(x = viability, y = efficiency_avg)) +
    geom_point(aes(color = well_position), size = 3) +
    geom_smooth(method = "lm", se = FALSE, color = "black") +
    ggpmisc::stat_poly_eq(
      aes(label = after_stat(rr.label)),
      formula = y ~ x,
      parse = TRUE,
      label.x.npc = "left",
      label.y.npc = "top"
    ) +
    facet_wrap(
      ~ donor_recruitment * promoter,
      labeller = labeller(donor_recruitment = c("LexA-FHA" = "LexA-FHA", "No" = ""))
    ) +
    labs(
      x = "Viability (relative to A09 control)",
      y = "Average efficiency (%)",
      title = paste("First timepoint viability:", cas_variant_name)
    ) +
    publication_theme(legend_position = "none", aspect_ratio = 1)
}
