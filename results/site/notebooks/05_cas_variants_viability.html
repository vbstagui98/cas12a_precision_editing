resolve_project_root <- function() {
  rstudio_path <- ""
  if (requireNamespace("rstudioapi", quietly = TRUE)) {
    rstudio_path <- tryCatch(rstudioapi::getSourceEditorContext()$path, error = function(e) "")
    if (!nzchar(rstudio_path)) {
      rstudio_path <- tryCatch(rstudioapi::getActiveDocumentContext()$path, error = function(e) "")
    }
  }

  search_roots <- unique(c(
    Sys.getenv("CAS12A_NOTEBOOK_PROJECT_ROOT", unset = ""),
    tryCatch(knitr::current_input(dir = TRUE), error = function(e) ""),
    rstudio_path,
    getwd(),
    normalizePath(file.path(getwd(), ".."), mustWork = FALSE),
    normalizePath(file.path(getwd(), "..", ".."), mustWork = FALSE)
  ))
  search_roots <- search_roots[nzchar(search_roots)]

  for (root in search_roots) {
    current <- normalizePath(root, mustWork = FALSE)
    if (!dir.exists(current)) {
      current <- dirname(current)
    }

    repeat {
      required_paths <- c(
        file.path(current, "_quarto.yml"),
        file.path(current, "config", "paths.R"),
        file.path(current, "R", "common.R")
      )

      if (all(file.exists(required_paths))) {
        return(current)
      }

      parent <- dirname(current)
      if (identical(parent, current)) {
        break
      }

      current <- parent
    }
  }

  stop(
    paste(
      "Could not locate the notebook project root.",
      "Save the notebook and either run it from the project root,",
      "or set Sys.setenv(CAS12A_NOTEBOOK_PROJECT_ROOT = '/path/to/New project')."
    ),
    call. = FALSE
  )
}

off_counts <- read_csv_clean(file.path(paths$cognate_dr_root, "colony_counts.csv")) %>%
  tidyr::separate(identifier, into = c("strain", "guide_plasmid"), sep = "_", remove = FALSE) %>%
  dplyr::mutate(
    var = dplyr::case_when(
      strain == "ys85" ~ "enAsCas12a",
      strain == "ys86" ~ "LbCas12a(CO)",
      strain == "ys87" ~ "FnCas12a(CO)",
      strain == "ys88" ~ "FnCas12a",
      strain == "ys89" ~ "LbCas12a",
      strain == "ys90" ~ "AsCas12a",
      strain == "ys91" ~ "MbCas12a",
      TRUE ~ strain
    ),
    rep = as.character(replicate)
  ) %>%
  dplyr::filter(guide_plasmid %in% c("p120", "p136"))

off_control_counts <- off_counts %>%
  dplyr::filter(guide_plasmid == "p120") %>%
  dplyr::group_by(var) %>%
  dplyr::summarise(control_count = mean(true_colony_count, na.rm = TRUE), .groups = "drop")

off_target_counts <- off_counts %>%
  dplyr::filter(guide_plasmid == "p136") %>%
  dplyr::select(var, rep, true_colony_count)

efficiency_off_data <- gfp_off %>%
  dplyr::left_join(off_target_counts, by = c("var", "rep")) %>%
  dplyr::left_join(off_control_counts, by = "var") %>%
  dplyr::mutate(viability = true_colony_count / control_count)

write_csv_checked(efficiency_off_data, file.path(tables_dir, "efficiency_off_data.csv"))
