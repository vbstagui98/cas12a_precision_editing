resolve_project_root <- function() {
  rstudio_path <- ""
  if (requireNamespace("rstudioapi", quietly = TRUE)) {
    rstudio_path <- tryCatch(rstudioapi::getSourceEditorContext()$path, error = function(e) "")
    if (!nzchar(rstudio_path)) {
      rstudio_path <- tryCatch(rstudioapi::getActiveDocumentContext()$path, error = function(e) "")
    }
  }

  search_roots <- unique(c(
    Sys.getenv("CAS12A_NOTEBOOK_PROJECT_ROOT", unset = ""),
    tryCatch(knitr::current_input(dir = TRUE), error = function(e) ""),
    rstudio_path,
    getwd(),
    normalizePath(file.path(getwd(), ".."), mustWork = FALSE),
    normalizePath(file.path(getwd(), "..", ".."), mustWork = FALSE)
  ))
  search_roots <- search_roots[nzchar(search_roots)]

  for (root in search_roots) {
    current <- normalizePath(root, mustWork = FALSE)
    if (!dir.exists(current)) {
      current <- dirname(current)
    }

    repeat {
      required_paths <- c(
        file.path(current, "_quarto.yml"),
        file.path(current, "config", "paths.R"),
        file.path(current, "R", "common.R")
      )

      if (all(file.exists(required_paths))) {
        return(current)
      }

      parent <- dirname(current)
      if (identical(parent, current)) {
        break
      }

      current <- parent
    }
  }

  stop(
    paste(
      "Could not locate the notebook project root.",
      "Save the notebook and either run it from the project root,",
      "or set Sys.setenv(CAS12A_NOTEBOOK_PROJECT_ROOT = '/path/to/New project')."
    ),
    call. = FALSE
  )
}

plot_amplicon_heatmaps <- function(data, fill_high = "#E69F00", col_max = 100) {
  timepoints <- unique(data$timepoint)
  output <- vector("list", length(timepoints))
  names(output) <- timepoints

  for (timepoint_name in timepoints) {
    timepoint_data <- data %>%
      dplyr::filter(timepoint == timepoint_name)

    stats_df <- timepoint_data %>%
      dplyr::group_by(condition_label) %>%
      dplyr::summarise(true_cond_avg = mean(avg_value, na.rm = TRUE), .groups = "drop")

    timepoint_data <- timepoint_data %>%
      dplyr::left_join(stats_df, by = "condition_label") %>%
      dplyr::mutate(y_label_final = paste0(condition_label, "\n(Avg: ", round(true_cond_avg, 1), "%)"))

    y_levels <- timepoint_data %>%
      dplyr::arrange(dplyr::desc(true_cond_avg)) %>%
      dplyr::pull(y_label_final) %>%
      unique()

    x_levels <- timepoint_data %>%
      dplyr::group_by(well_position) %>%
      dplyr::summarise(mean_eff = mean(avg_value, na.rm = TRUE), .groups = "drop") %>%
      dplyr::arrange(mean_eff) %>%
      dplyr::pull(well_position)

    timepoint_data$y_label_final <- factor(timepoint_data$y_label_final, levels = y_levels)
    timepoint_data$well_position <- factor(timepoint_data$well_position, levels = x_levels)

    output[[timepoint_name]] <- ggplot(timepoint_data, aes(x = well_position, y = y_label_final, fill = avg_value)) +
      geom_tile(color = "white", linewidth = 0.6) +
      scale_fill_gradient(low = "#F8F9FA", high = fill_high, limits = c(0, col_max), oob = scales::squish) +
      coord_fixed() +
      labs(x = "Guide", y = NULL, title = paste("Amplicon HDR heatmap", timepoint_name)) +
      theme_minimal(base_size = 11) +
      theme(
        panel.grid = element_blank(),
        axis.text.x = element_text(angle = 45, vjust = 1, hjust = 1, color = "black"),
        axis.text.y = element_text(face = "bold", color = "black", size = 8),
        plot.title = element_text(face = "bold", size = 12, hjust = 0)
      )
  }

  output
}
