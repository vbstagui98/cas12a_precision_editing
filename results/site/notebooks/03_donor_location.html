resolve_project_root <- function() {
  rstudio_path <- ""
  if (requireNamespace("rstudioapi", quietly = TRUE)) {
    rstudio_path <- tryCatch(rstudioapi::getSourceEditorContext()$path, error = function(e) "")
    if (!nzchar(rstudio_path)) {
      rstudio_path <- tryCatch(rstudioapi::getActiveDocumentContext()$path, error = function(e) "")
    }
  }

  search_roots <- unique(c(
    Sys.getenv("CAS12A_NOTEBOOK_PROJECT_ROOT", unset = ""),
    tryCatch(knitr::current_input(dir = TRUE), error = function(e) ""),
    rstudio_path,
    getwd(),
    normalizePath(file.path(getwd(), ".."), mustWork = FALSE),
    normalizePath(file.path(getwd(), "..", ".."), mustWork = FALSE)
  ))
  search_roots <- search_roots[nzchar(search_roots)]

  for (root in search_roots) {
    current <- normalizePath(root, mustWork = FALSE)
    if (!dir.exists(current)) {
      current <- dirname(current)
    }

    repeat {
      required_paths <- c(
        file.path(current, "_quarto.yml"),
        file.path(current, "config", "paths.R"),
        file.path(current, "R", "common.R")
      )

      if (all(file.exists(required_paths))) {
        return(current)
      }

      parent <- dirname(current)
      if (identical(parent, current)) {
        break
      }

      current <- parent
    }
  }

  stop(
    paste(
      "Could not locate the notebook project root.",
      "Save the notebook and either run it from the project root,",
      "or set Sys.setenv(CAS12A_NOTEBOOK_PROJECT_ROOT = '/path/to/New project')."
    ),
    call. = FALSE
  )
}

standardize_donor_table <- function(data, nuisance_column) {
  data <- data %>%
    dplyr::select(-1, -dplyr::any_of(nuisance_column))

  if (ncol(data) != 7) {
    stop("Unexpected donor table layout after dropping nuisance columns.", call. = FALSE)
  }

  names(data) <- c("dist", "replicate", "frc_alt", "frc_ref", "hdr_pct", "timepoint", "strain")

  data %>%
    dplyr::mutate(
      dist = as.numeric(dist),
      replicate = as.character(replicate)
    )
}

plot_flexible_final <- function(data, selected_timepoints, x_var, y_var, x_label, y_limits = c(0, 100)) {
  lookup <- tibble::tibble(
    cas_variant = names(selected_timepoints),
    selected_timepoint = unname(selected_timepoints)
  )

  plot_data_raw <- data %>%
    dplyr::left_join(lookup, by = "cas_variant") %>%
    dplyr::filter(timepoint == selected_timepoint)

  if (identical(x_var, "dist")) {
    plot_data_raw <- plot_data_raw %>%
      dplyr::filter(as.numeric(.data[[x_var]]) != 6)
  }

  plot_data_avg <- plot_data_raw %>%
    dplyr::group_by(cas_variant, .data[[x_var]]) %>%
    dplyr::summarise(avg_y = mean(.data[[y_var]], na.rm = TRUE), .groups = "drop")

  variant_colors <- c("enAsCas12a" = "#E69F00", "FnCas12a" = "#009E73")
  x_values <- sort(unique(plot_data_raw[[x_var]]))
  x_limits <- c(min(x_values) - 0.5, max(x_values) + 0.5)

  p_line <- ggplot() +
    geom_line(
      data = plot_data_avg,
      aes(x = .data[[x_var]], y = avg_y, color = cas_variant, group = cas_variant),
      linewidth = 0.8
    ) +
    geom_point(
      data = plot_data_raw,
      aes(x = .data[[x_var]], y = .data[[y_var]], color = cas_variant),
      size = 1,
      alpha = 0.8,
      position = position_dodge(width = 0.3)
    ) +
    scale_color_manual(values = variant_colors) +
    scale_x_continuous(breaks = x_values, limits = x_limits, expand = c(0, 0)) +
    scale_y_continuous(limits = y_limits, expand = expansion(mult = c(0.02, 0.1))) +
    labs(
      title = paste(names(selected_timepoints), selected_timepoints, sep = ": ", collapse = ", "),
      y = "Percent (%)",
      x = NULL
    ) +
    theme_minimal(base_size = 10) +
    theme(
      panel.grid = element_blank(),
      axis.text.x = element_blank(),
      legend.position = "none",
      panel.border = element_rect(colour = "black", fill = NA, linewidth = 0.5),
      strip.background = element_blank()
    )

  p_heat <- ggplot(plot_data_avg, aes(x = .data[[x_var]], y = cas_variant)) +
    geom_tile(aes(fill = cas_variant, alpha = avg_y), color = "white", linewidth = 0.5) +
    scale_fill_manual(values = variant_colors) +
    scale_alpha_continuous(range = c(0.1, 1), limits = y_limits, name = "Avg %") +
    scale_x_continuous(breaks = x_values, limits = x_limits, expand = c(0, 0)) +
    coord_fixed(ratio = 1) +
    labs(x = x_label, y = NULL) +
    theme_minimal(base_size = 14) +
    theme(
      panel.grid = element_blank(),
      axis.text.y = element_text(face = "bold"),
      legend.position = "none"
    )

  (p_line / p_heat) +
    patchwork::plot_layout(heights = c(1, 1), guides = "collect") &
    theme(plot.margin = margin(1, 1, 1, 1))
}

plot_short_guide_viability <- function(data, strain_name, high_color) {
  data %>%
    dplyr::filter(strain == strain_name) %>%
    ggplot(aes(x = guide_length, y = viability)) +
    geom_point(aes(color = editing_efficiency * 100)) +
    scale_color_gradient(
      low = "#999999",
      high = high_color,
      limits = c(0, 100),
      oob = scales::squish,
      name = "HDR fraction %"
    ) +
    coord_cartesian(ylim = c(0, 1)) +
    labs(x = "Guide length", y = "Viability") +
    publication_theme(legend_position = "right", aspect_ratio = 1)
}
